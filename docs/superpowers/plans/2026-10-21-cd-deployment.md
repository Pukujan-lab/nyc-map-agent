# CD Deployment Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Push to `main` → CI passes → gravebuster builds and swaps the running nyc-map-agent container automatically, with a password-protected dashboard at `nyc.design-bakery.com/deploy`.

**Architecture:** Three containers on gravebuster in one compose project. `nyc-map-web` runs the app; `nyc-map-cd` is a small Hono service holding `docker.sock` that runs the deploy pipeline; `nyc-map-edge` is Caddy, which routes `/deploy/api/*` to the CD service, `/deploy*` to the ABA-generated dashboard's static files, and everything else to the app. Cloudflare Tunnel reaches only the edge container.

**Tech Stack:** TypeScript, Hono + `@hono/node-server`, Vitest, Docker Compose, Caddy 2, GitHub Actions. The dashboard is generated by app-builder-automation (React + Vite + shadcn) and committed to `deploy/dashboard/`.

**Spec:** `docs/superpowers/specs/2026-10-21-cd-deployment-design.md`

## Global Constraints

- Node 22 in all images (`node:22-slim`); the repo's `packageManager` is `pnpm@11.19.0`.
- `deploy/cd/` uses only dependencies already in the root `package.json` (`hono`, `@hono/node-server`, `zod`; `tsx` and `vitest` are dev dependencies). Do not add packages.
- Every new `.ts` file under `deploy/` must pass `pnpm typecheck`, `pnpm lint`, `pnpm knip`, and `pnpm test`.
- `deploy/dashboard/` is a separate package: exclude it from the root `tsconfig.json`, `knip.json`, and ESLint.
- Secrets never enter the repo. They live in `/srv/nyc-map-agent/.env` (mode 600) on the server.
- The map app stays public; only `/deploy/api/*` is authenticated.
- Deploy routes return `202` immediately; nothing waits inline on a build (Cloudflare drops requests near 100s).
- Run ids are `YYYYMMDDTHHMMSSZ-<sha12>`.
- No `curl` in `node:22-slim`; health checks use `node -e`.

```ts
import { appendFileSync, existsSync, mkdirSync, readFileSync } from "node:fs";
import { dirname } from "node:path";
import type { Run } from "./types";

/** Appends one JSON line per run record. Later records for the same id win. */
export function appendRun(file: string, run: Run): void {
  mkdirSync(dirname(file), { recursive: true });
  appendFileSync(file, `${JSON.stringify(run)}\n`);
}

export function readRuns(file: string, limit = 20): Run[] {
  if (!existsSync(file)) return [];
  const byId = new Map<string, Run>();
  for (const line of readFileSync(file, "utf8").split("\n")) {
    if (!line.trim()) continue;
    try {
      const run = JSON.parse(line) as Run;
      if (run && typeof run.id === "string") byId.set(run.id, run);
    } catch {
      // A truncated line from a killed process; the next record still reads.
    }
  }
  return [...byId.values()]
    .sort((a, b) => (a.startedAt < b.startedAt ? 1 : a.startedAt > b.startedAt ? -1 : 0))
    .slice(0, limit);
}

export function currentRun(file: string): Run | null {
  return readRuns(file, Number.MAX_SAFE_INTEGER).find((r) => r.status === "succeeded" || r.status === "rolled-back") ?? null;
}

/** A run left `running` by a killed CD container is closed on the next start. */
export function markInterrupted(file: string, now = new Date()): number {
  const open = readRuns(file, Number.MAX_SAFE_INTEGER).filter((r) => r.status === "running");
  for (const run of open) {
    appendRun(file, {
      ...run,
      status: "interrupted",
      finishedAt: now.toISOString(),
      durationMs: now.getTime() - Date.parse(run.startedAt),
      error: "the deploy server restarted mid-run",
    });
  }
  return open.length;
}
```

```ts
import { createHash, createHmac, timingSafeEqual } from "node:crypto";

export const SESSION_COOKIE = "nyc_cd_session";
const SESSION_TTL_MS = 7 * 24 * 60 * 60 * 1000;

function hmac(secret: string, payload: string): string {
  return createHmac("sha256", secret).update(payload).digest("base64url");
}

export function sessionExpiry(nowMs = Date.now()): number {
  return nowMs + SESSION_TTL_MS;
}

export function signSession(secret: string, expMs: number): string {
  const payload = String(expMs);
  return `${payload}.${hmac(secret, payload)}`;
}

export function verifySession(secret: string, cookie: string | undefined, nowMs = Date.now()): boolean {
  if (!cookie) return false;
  const parts = cookie.split(".");
  if (parts.length !== 2 || !parts[0] || !parts[1]) return false;
  const [payload, sig] = parts;
  const expected = Buffer.from(hmac(secret, payload));
  const actual = Buffer.from(sig);
  if (expected.length !== actual.length || !timingSafeEqual(expected, actual)) return false;
  const exp = Number(payload);
  return Number.isFinite(exp) && exp > nowMs;
}

/** Compares SHA-256 digests so differing input lengths never throw. */
export function checkPassword(input: string, expected: string): boolean {
  const a = createHash("sha256").update(input).digest();
  const b = createHash("sha256").update(expected).digest();
  return timingSafeEqual(a, b);
}

export function checkBearer(header: string | undefined, token: string): boolean {
  const match = header?.match(/^Bearer (.+)$/);
  return match ? checkPassword(match[1], token) : false;
}

export interface RateLimiter {
  blocked(ip: string, nowMs?: number): boolean;
  record(ip: string, nowMs?: number): void;
}

export function createRateLimiter(opts: { max?: number; windowMs?: number } = {}): RateLimiter {
  const max = opts.max ?? 5;
  const windowMs = opts.windowMs ?? 15 * 60 * 1000;
  const hits = new Map<string, number[]>();
  const recent = (ip: string, nowMs: number) => (hits.get(ip) ?? []).filter((t) => nowMs - t < windowMs);
  return {
    blocked(ip, nowMs = Date.now()) {
      const arr = recent(ip, nowMs);
      hits.set(ip, arr);
      return arr.length >= max;
    },
    record(ip, nowMs = Date.now()) {
      hits.set(ip, [...recent(ip, nowMs), nowMs]);
    },
  };
}
```

### Task 5: Deploy lock

**Files:**
- Create: `deploy/cd/lock.ts`
- Test: `deploy/cd/lock.test.ts`

**Interfaces:**
- Produces:
  - `acquireLock(file: string, opts?: { nowMs?: number; staleMs?: number }): boolean`
  - `releaseLock(file: string): void`
  - `lockOwner(file: string): { pid: number; at: number } | null`

The lock is a **file** at `config.lockDir` (`<state>/deploy.lock`), published
atomically with `linkSync` from a temp file whose content is already written.
`link` cannot publish an inode without its content, so no concurrent acquirer
can ever observe a lock that exists but has no owner — which is what makes a
single `mkdir`-then-`writeFileSync` sequence unsound.

```ts
  it("acquires an unheld lock and records the owner", () => {

    expect(acquireLock(file, { nowMs: NOW })).toBe(true);
```

```ts
import { linkSync, mkdirSync, readFileSync, rmSync, writeFileSync } from "node:fs";
import { dirname } from "node:path";

export interface LockOwner {
  pid: number;
  at: number;
}

export function lockOwner(file: string): LockOwner | null {
  try {
    const owner = JSON.parse(readFileSync(file, "utf8")) as LockOwner;
    return typeof owner.at === "number" && typeof owner.pid === "number" ? owner : null;
  } catch {
    return null;
  }
}

/**
 * `link` publishes an inode whose content is already written, so a reader can
 * never observe a lock that exists but has no owner. The path is the lock.
 */
export function acquireLock(file: string, opts: { nowMs?: number; staleMs?: number } = {}): boolean {
  const nowMs = opts.nowMs ?? Date.now();
  const staleMs = opts.staleMs ?? 30 * 60 * 1000;
  // The state directory does not exist on a fresh host; creating it here keeps
  // the first deploy from failing every acquire with ENOENT.
  mkdirSync(dirname(file), { recursive: true });

  const tmp = `${file}.${process.pid}.tmp`;
  writeFileSync(tmp, JSON.stringify({ pid: process.pid, at: nowMs }));
  try {
    try {
      linkSync(tmp, file);
      return true;
    } catch (err) {
      if ((err as NodeJS.ErrnoException).code !== "EEXIST") throw err;
    }
    const owner = lockOwner(file);
    if (owner && nowMs - owner.at < staleMs) return false;
    // Unreadable content cannot come from a concurrent creator — `link`
    // publishes complete content — so it is a damaged lock and safe to clear.
    rmSync(file, { force: true });
    try {
      linkSync(tmp, file);
      return true;
    } catch (err) {
      if ((err as NodeJS.ErrnoException).code !== "EEXIST") throw err;
      return false;
    }
  } finally {
    rmSync(tmp, { force: true });
  }
}

/** Removes the lock only when this process owns it. */
export function releaseLock(file: string): void {
  if (lockOwner(file)?.pid !== process.pid) return;
  rmSync(file, { force: true });
}
```

```ts
import { spawn } from "node:child_process";

export interface RunOptions {
  cwd?: string;
  env?: Record<string, string>;
  onLine?: (line: string) => void;
}

export interface RunResult {
  code: number;
  output: string;
}

export interface Runner {
  run(cmd: string, args: string[], opts?: RunOptions): Promise<RunResult>;
}

interface LineSplitter {
  push(text: string): void;
  flush(): void;
}

/** The only place the deploy code shells out. Tests inject a fake Runner. */
export function createRunner(): Runner {
  return {
    run(cmd, args, opts = {}) {
      return new Promise<RunResult>((resolve, reject) => {
        const child = spawn(cmd, args, {
          cwd: opts.cwd,
          env: { ...process.env, ...opts.env },
          shell: false,
        });
        let output = "";

        // Each stream needs its own partial line: a stdout fragment followed by
        // a stderr line must not be glued into a line neither stream wrote.
        const splitter = (): LineSplitter => {
          let pending = "";
          const emit = (line: string) => opts.onLine?.(line.endsWith("\r") ? line.slice(0, -1) : line);
          return {
            push(text) {
              pending += text;
              const lines = pending.split("\n");
              pending = lines.pop() ?? "";
              for (const line of lines) emit(line);
            },
            flush() {
              if (pending) emit(pending);
              pending = "";
            },
          };
        };
        const out = splitter();
        const err = splitter();

        // Decoding at the stream keeps a multi-byte character split across two
        // chunks from being emitted as a replacement character.
        child.stdout.setEncoding("utf8");
        child.stderr.setEncoding("utf8");
        child.stdout.on("data", (text: string) => {
          output += text;
          out.push(text);
        });
        child.stderr.on("data", (text: string) => {
          output += text;
          err.push(text);
        });
        child.on("error", reject);
        child.on("close", (code) => {
          out.flush();
          err.flush();
          resolve({ code: code ?? -1, output });
        });
      });
    },
  };
}
```

```ts
import { appendRun, currentRun } from "./state";
import { acquireLock, releaseLock } from "./lock";
import type { Runner } from "./runner";
import { emptySteps, type Run, type Step, type StepName } from "./types";

export interface PipelineConfig {
  repoDir: string;
  deployDir: string;
  stateFile: string;
  lockDir: string;
  envFile: string;
  buildArgs: Record<string, string>;
  healthTimeoutMs?: number;
  healthPollMs?: number;
}

export interface PipelineDeps {
  runner: Runner;
  now?: () => Date;
  sleep?: (ms: number) => Promise<void>;
  onLog?: (runId: string, line: string) => void;
  /** Called on every state change, including the first one after the sha resolves. */
  onUpdate?: (run: Run) => void;
}

export type DeployRequest = { kind: "deploy"; sha?: string } | { kind: "rollback" };

export class BusyError extends Error {
  constructor() {
    super("a deploy is already running");
    this.name = "BusyError";
  }
}

const CONTAINER = "nyc-map-web";
const ZERO_SHA = "0".repeat(40);

export function runId(date: Date, sha: string): string {
  const stamp = date.toISOString().replace(/[-:]/g, "").replace(/\.\d{3}Z$/, "Z");
  return `${stamp}-${sha.slice(0, 12)}`;
}

export async function startDeploy(deps: PipelineDeps, config: PipelineConfig, request: DeployRequest): Promise<Run> {
  const now = deps.now ?? (() => new Date());
  const sleep = deps.sleep ?? ((ms: number) => new Promise((r) => setTimeout(r, ms)));
  const { runner } = deps;

  if (request.kind === "deploy" && request.sha && !/^[0-9a-f]{40}$/.test(request.sha)) {
    throw new Error("sha must be a full 40-character sha");
  }
  if (!acquireLock(config.lockDir, { nowMs: now().getTime() })) throw new BusyError();

  const startedAt = now();
  // A deploy's id carries its sha, known up front when CI supplies one and
  // resolved from origin/main otherwise. A rollback's sha is the commit the last
  // good deploy replaced, read from the state file once the lock is held.
  const provisionalSha = request.kind === "deploy" ? (request.sha ?? ZERO_SHA) : ZERO_SHA;
  let run: Run = {
    id: runId(startedAt, provisionalSha),
    sha: provisionalSha,
    kind: request.kind,
    status: "running",
    startedAt: startedAt.toISOString(),
    finishedAt: null,
    durationMs: null,
    image: "",
    previousImage: null,
    steps: emptySteps(),
    error: null,
  };

  const lines: string[] = [];
  const log = (line: string) => {
    lines.push(line);
    deps.onLog?.(run.id, line);
  };
  const publish = (next: Run): Run => {
    const previousId = run.id;
    run = next;
    // Until the sha is known the record has a placeholder id; persisting it would
    // leave a phantom "running" run in the history under a second id. Once the sha
    // is known — supplied by CI or resolved from origin/main — every transition is
    // persisted, so /status lists the run while it is still in progress. A terminal
    // record is always written, so a fetch failure is not lost.
    const provisional = run.sha === ZERO_SHA && run.status === "running";
    if (!provisional) appendRun(config.stateFile, run);
    // Lines logged before the id was known were keyed to the placeholder; replay
    // them under the real id so the SSE stream replays completely.
    if (run.id !== previousId) for (const line of lines) deps.onLog?.(run.id, line);
    deps.onUpdate?.(run);
    return run;
  };
  const step = (name: StepName, status: Step["status"]): Run => {
    const at = now();
    return publish({
      ...run,
      steps: run.steps.map((s) =>
        s.name === name
          ? {
              ...s,
              status,
              startedAt: status === "running" ? at.toISOString() : s.startedAt,
              finishedAt: status === "succeeded" || status === "failed" ? at.toISOString() : s.finishedAt,
            }
          : s,
      ),
    });
  };
  const finish = (status: Run["status"], error: string | null): Run => {
    const ended = now();
    return publish({
      ...run,
      status,
      error,
      finishedAt: ended.toISOString(),
      durationMs: ended.getTime() - startedAt.getTime(),
      // A run that stops early leaves later steps untouched; "skipped" is what the
      // dashboard renders for them, so mark them here rather than at every exit.
      steps: run.steps.map((s) => (s.status === "pending" ? { ...s, status: "skipped" } : s)),
    });
  };

  publish(run);
  try {
    // fetch
    step("fetch", "running");
    log("git fetch origin");
    const fetched = await runner.run("git", ["fetch", "--prune", "origin", "main"], { cwd: config.repoDir, onLine: log });
    if (fetched.code !== 0) {
      step("fetch", "failed");
      return finish("failed", "git fetch failed");
    }
    if (request.kind === "deploy") {
      const resolved = await runner.run(
        "git",
        ["rev-parse", request.sha ? `${request.sha}^{commit}` : "origin/main^{commit}"],
        { cwd: config.repoDir, onLine: log },
      );
      const sha = resolved.output.trim().split("\n").pop() ?? "";
      if (resolved.code !== 0 || !/^[0-9a-f]{40}$/.test(sha)) {
        step("fetch", "failed");
        return finish("failed", "could not resolve the requested ref");
      }
      // The id carries the sha, so it is published once here — the server answers
      // the HTTP request on this update.
      publish({ ...run, id: runId(startedAt, sha), sha });
    }
    step("fetch", "succeeded");

    // The image running right now: what an automatic swap-back restores, and what
    // a manual rollback is moving away from.
    const current = await runner.run("docker", ["inspect", CONTAINER, "--format", "{{.Config.Image}}"], { onLine: log });
    const liveImage = current.code === 0 ? current.output.trim() || null : null;

    // A manual rollback restores the image the last good deploy replaced. That run
    // carries both the sha (so this run is identifiable) and the replaced image.
    const previous = request.kind === "rollback" ? currentRun(config.stateFile) : null;
    publish({
      ...run,
      previousImage: liveImage,
      ...(previous ? { sha: previous.sha, id: runId(startedAt, previous.sha) } : {}),
    });
    log(`live image: ${liveImage ?? "unknown"}`);

    const targetImage =
      request.kind === "rollback" ? (previous?.previousImage ?? "") : `nyc-map-agent:${run.sha.slice(0, 12)}`;

    if (request.kind === "rollback" && !targetImage) {
      return finish("failed", "no previous image to roll back to");
    }

    if (request.kind === "deploy") {
      // checkout
      step("checkout", "running");
      const checkout = await runner.run("git", ["checkout", "--detach", run.sha], { cwd: config.repoDir, onLine: log });
      if (checkout.code !== 0) {
        step("checkout", "failed");
        return finish("failed", "git checkout failed");
      }
      const clean = await runner.run("git", ["clean", "-xdf"], { cwd: config.repoDir, onLine: log });
      if (clean.code !== 0) {
        step("checkout", "failed");
        return finish("failed", "git clean failed");
      }
      step("checkout", "succeeded");

      // build
      step("build", "running");
      const buildArgs = Object.entries(config.buildArgs).flatMap(([k, v]) => ["--build-arg", `${k}=${v}`]);
      const built = await runner.run("docker", ["build", "-t", targetImage, ...buildArgs, "."], {
        cwd: config.repoDir,
        onLine: log,
      });
      if (built.code !== 0) {
        step("build", "failed");
        return finish("failed", "docker build failed");
      }
      step("build", "succeeded");
    } else {
      step("checkout", "skipped");
      step("build", "skipped");
    }

    // swap
    step("swap", "running");
    if (!(await swap(config, runner, targetImage, log))) {
      step("swap", "failed");
      return finish("failed", "docker compose up failed");
    }
    publish({ ...run, image: targetImage });
    step("swap", "succeeded");

    // health
    step("health", "running");
    if (!(await waitHealthy(config, runner, sleep, log))) {
      step("health", "failed");
      log(`health check failed; rolling back to ${run.previousImage ?? "nothing"}`);
      if (run.previousImage && run.previousImage !== targetImage) {
        await swap(config, runner, run.previousImage, log);
        await waitHealthy(config, runner, sleep, log);
      }
      return finish("failed", "the new container never became healthy");
    }
    step("health", "succeeded");

    return finish(request.kind === "rollback" ? "rolled-back" : "succeeded", null);
  } finally {
    releaseLock(config.lockDir);
  }
}

async function swap(config: PipelineConfig, runner: Runner, image: string, onLine: (line: string) => void): Promise<boolean> {
  const result = await runner.run(
    "docker",
    [
      "compose",
      "-f",
      `${config.deployDir}/docker-compose.yml`,
      "--project-directory",
      config.deployDir,
      "--env-file",
      config.envFile,
      "up",
      "-d",
      "--no-deps",
      "web",
    ],
    { cwd: config.deployDir, env: { NYC_MAP_IMAGE: image }, onLine },
  );
  return result.code === 0;
}

async function waitHealthy(
  config: PipelineConfig,
  runner: Runner,
  sleep: (ms: number) => Promise<void>,
  onLine: (line: string) => void,
): Promise<boolean> {
  const timeoutMs = config.healthTimeoutMs ?? 90_000;
  const pollMs = config.healthPollMs ?? 2_000;
  const deadline = Date.now() + timeoutMs;
  while (Date.now() < deadline) {
    const result = await runner.run("docker", ["inspect", CONTAINER, "--format", "{{.State.Health.Status}}"], { onLine });
    const status = result.output.trim();
    if (result.code === 0 && status === "healthy") return true;
    if (status === "unhealthy") return false;
    await sleep(pollMs);
  }
  return false;
}
```

```ts
import { join } from "node:path";
```

```ts
  checkPassword,

  signSession,
  SESSION_COOKIE,
  verifySession,
} from "./auth";
import { BusyError, startDeploy, type PipelineConfig } from "./pipeline";
import { createRunner, type Runner } from "./runner";
import { currentRun, markInterrupted, readRuns } from "./state";
import type { Run } from "./types";

export interface ServerConfig {
  password: string;
  sessionSecret: string;
  deployToken: string;
  stateFile: string;
  pipeline: PipelineConfig;
  now?: () => Date;

export interface LogStore {
  append(runId: string, line: string): void;
  lines(runId: string): string[];
  waiters(runId: string): Set<(line: string) => void>;
  done(runId: string): void;
  isDone(runId: string): boolean;
}

export function createLogStore(cap = 5_000): LogStore {
  const buffers = new Map<string, string[]>();
  const listeners = new Map<string, Set<(line: string) => void>>();
  const finished = new Set<string>();
  return {
    append(runId, line) {
      const buf = buffers.get(runId) ?? [];
      buf.push(line);
      if (buf.length > cap) buf.splice(0, buf.length - cap);
      buffers.set(runId, buf);
      for (const fn of listeners.get(runId) ?? []) fn(line);
    },
    lines: (runId) => buffers.get(runId) ?? [],
    waiters(runId) {
      let set = listeners.get(runId);
      if (!set) {
        set = new Set();
        listeners.set(runId, set);
      }
      return set;
    },
    done: (runId) => void finished.add(runId),
    isDone: (runId) => finished.has(runId),
  };
}

function readCookie(header: string | undefined, name: string): string | undefined {
  for (const part of (header ?? "").split(";")) {
    const [k, ...rest] = part.trim().split("=");
    if (k === name) return rest.join("=");
  }
  return undefined;
}

export function createApp(config: ServerConfig, deps: { runner: Runner }): Hono {
  const app = new Hono();
  const limiter = createRateLimiter();
  const logs = createLogStore();
  const now = config.now ?? (() => new Date());

  const authed = (c: { req: { header: (k: string) => string | undefined } }) =>
    verifySession(config.sessionSecret, readCookie(c.req.header("cookie"), SESSION_COOKIE), now().getTime());

  // CI starts a deploy with a bearer token, then has no cookie to poll with, so
  // reading a single run accepts either credential. Status and logs stay
  // session-only.
  const authedOrBearer = (c: { req: { header: (k: string) => string | undefined } }) =>
    authed(c) || checkBearer(c.req.header("authorization"), config.deployToken);

  const unauthorized = (c: { json: (b: unknown, s: 401) => Response }) => c.json({ error: "unauthorized" }, 401);

  app.post("/deploy/api/login", async (c) => {
    const ip = c.req.header("x-forwarded-for")?.split(",")[0].trim() || "local";
    if (limiter.blocked(ip, now().getTime())) return c.json({ error: "rate-limited" }, 429);
    const body = (await c.req.json().catch(() => ({}))) as { password?: string };
    if (!body.password || !checkPassword(body.password, config.password)) {
      limiter.record(ip, now().getTime());
      return c.json({ error: "invalid" }, 401);
    }
    const exp = sessionExpiry(now().getTime());
    const cookie = `${SESSION_COOKIE}=${signSession(config.sessionSecret, exp)}; Path=/deploy; HttpOnly; Secure; SameSite=Lax; Max-Age=${Math.floor((exp - now().getTime()) / 1000)}`;
    c.header("set-cookie", cookie);
    return c.json({ ok: true });
  });

  app.post("/deploy/api/logout", (c) => {
    c.header("set-cookie", `${SESSION_COOKIE}=; Path=/deploy; HttpOnly; Secure; SameSite=Lax; Max-Age=0`);
    return c.json({ ok: true });
  });

  app.get("/deploy/api/status", async (c) => {
    if (!authed(c)) return unauthorized(c);
    const runs = readRuns(config.stateFile, 20);
    const live = currentRun(config.stateFile);
    let health: "healthy" | "unhealthy" | "starting" | "unknown" = "unknown";
    let subject = "";
    if (live) {
      const inspected = await deps.runner.run("docker", ["inspect", "nyc-map-web", "--format", "{{.State.Health.Status}}"]);
      const status = inspected.output.trim();
      health = status === "healthy" || status === "unhealthy" || status === "starting" ? status : "unknown";
      const show = await deps.runner.run("git", ["show", "-s", "--format=%s", live.sha], { cwd: config.pipeline.repoDir });
      subject = show.code === 0 ? show.output.trim() : "";
    }
    return c.json({
      current: live
        ? { sha: live.sha, subject, image: live.image, health, deployedAt: live.finishedAt ?? live.startedAt }
        : null,
      running: runs.find((r) => r.status === "running") ?? null,
      runs,
    });
  });

  app.get("/deploy/api/deploys/:id", (c) => {
    if (!authedOrBearer(c)) return unauthorized(c);
    const run = readRuns(config.stateFile, Number.MAX_SAFE_INTEGER).find((r) => r.id === c.req.param("id"));
    return run ? c.json(run) : c.json({ error: "not_found" }, 404);
  });

  app.get("/deploy/api/deploys/:id/log", (c) => {
    if (!authed(c)) return unauthorized(c);
    const id = c.req.param("id");
    return streamSSE(c, async (stream) => {
      for (const line of logs.lines(id)) await stream.writeSSE({ data: JSON.stringify({ line, at: now().toISOString() }) });
      if (logs.isDone(id)) return stream.writeSSE({ event: "done", data: "{}" });
      await new Promise<void>((resolve) => {
        const set = logs.waiters(id);
        const push = (line: string) => {
          void stream.writeSSE({ data: JSON.stringify({ line, at: now().toISOString() }) });
        };
        set.add(push);
        const timer = setInterval(() => {
          if (logs.isDone(id)) {
            clearInterval(timer);
            set.delete(push);
            void stream.writeSSE({ event: "done", data: "{}" }).then(() => resolve());
          }
        }, 500);
        stream.onAbort(() => {
          clearInterval(timer);
          set.delete(push);
          resolve();
        });
      });
    });
  });

  const start = (kind: "deploy" | "rollback", sha?: string): Promise<string> => {
    const firstUpdate = new Promise<string>((resolve) => {
      let resolved = false;
      const onUpdate = (run: Run) => {
        if (!resolved && run.sha !== "0".repeat(40)) {
          resolved = true;
          resolve(run.id);
        }
      };
      void startDeploy(
        {
          runner: deps.runner,
          now,
          onLog: (runId, line) => logs.append(runId, line),
          onUpdate,
        },
        config.pipeline,
        kind === "deploy" ? { kind, sha } : { kind },
      )
        .then((run) => {
          logs.done(run.id);
          if (!resolved) resolve(run.id);
        })
        .catch((err: unknown) => {
          // The pipeline throws before publishing only for a bad request or a busy lock;
          // both are handled by the caller before this point.
          logs.append("error", `error: ${String(err)}`);
          if (!resolved) resolve("");
        });
    });
    return firstUpdate;
  };

  app.post("/deploy/api/deploy", async (c) => {
    const bearer = checkBearer(c.req.header("authorization"), config.deployToken);
    if (!bearer && !authed(c)) return unauthorized(c);
    const body = (await c.req.json().catch(() => ({}))) as { sha?: string };
    if (body.sha && !/^[0-9a-f]{40}$/.test(body.sha)) return c.json({ error: "invalid_sha" }, 400);
    try {
      const id = await start("deploy", body.sha);
      if (!id) return c.json({ error: "busy" }, 409);
      return c.json({ id }, 202);
    } catch (err) {
      if (err instanceof BusyError) return c.json({ error: "busy" }, 409);
      throw err;
    }
  });

  app.post("/deploy/api/rollback", async (c) => {
    const bearer = checkBearer(c.req.header("authorization"), config.deployToken);
    if (!bearer && !authed(c)) return unauthorized(c);
    try {
      const id = await start("rollback");
      if (!id) return c.json({ error: "busy" }, 409);
      return c.json({ id }, 202);
    } catch (err) {
      if (err instanceof BusyError) return c.json({ error: "busy" }, 409);
      throw err;
    }
  });

  return app;
}
```

### Task 11: CI deploy job

**Files:**
- Modify: `.github/workflows/ci.yml`

**Interfaces:**
- Consumes: `POST /deploy/api/deploy` and `GET /deploy/api/deploys/:id`.
- Produces: automatic deploys on green `main`.

- [ ] **Step 1: Append the job**

```yaml
  deploy:
    needs: [check, size, deps, lint, api, e2e]
    # Rollout guard: Task 13 removes this line once the stack is bootstrapped.
    if: false
    runs-on: ubuntu-latest
    timeout-minutes: 30
    concurrency:
      group: deploy-main
      cancel-in-progress: false
    steps:
      - name: Trigger deploy
        id: trigger
        env:
          TOKEN: ${{ secrets.NYC_DEPLOY_TOKEN }}
        run: |
          set -euo pipefail
          response=$(curl -fsS -X POST \
            -H "Authorization: Bearer $TOKEN" \
            -H "Content-Type: application/json" \
            -d "{\"sha\":\"${{ github.sha }}\"}" \
            https://nyc.design-bakery.com/deploy/api/deploy)
          id=$(printf '%s' "$response" | python3 -c 'import json,sys; print(json.load(sys.stdin)["id"])')
          echo "id=$id" >> "$GITHUB_OUTPUT"
          echo "started run $id"
      - name: Wait for the run
        env:
          TOKEN: ${{ secrets.NYC_DEPLOY_TOKEN }}
          RUN_ID: ${{ steps.trigger.outputs.id }}
        run: |
          set -euo pipefail
          for i in $(seq 1 120); do
            body=$(curl -fsS -H "Authorization: Bearer $TOKEN" \
              "https://nyc.design-bakery.com/deploy/api/deploys/$RUN_ID" || true)
            status=$(printf '%s' "$body" | python3 -c 'import json,sys; d=json.load(sys.stdin); print(d.get("status",""))' 2>/dev/null || echo "")
            echo "run $RUN_ID: ${status:-pending}"
            case "$status" in
              succeeded|rolled-back) exit 0 ;;
              failed|interrupted)
                echo "deploy $status"
                curl -fsS -H "Authorization: Bearer $TOKEN" \
                  "https://nyc.design-bakery.com/deploy/api/deploys/$RUN_ID" | python3 -m json.tool
                exit 1 ;;
            esac
            sleep 10
          done
          echo "timed out waiting for the deploy"; exit 1
```

**Type consistency:** `Run`/`Step`/`Status` are defined once in `types.ts` and imported everywhere. `runId` (pipeline) is the only id generator; the server derives the id from the pipeline's `onUpdate` callback rather than recomputing it. `startDeploy` always releases the lock in `finally`. `BusyError` is the only thrown control-flow error and maps to `409`.

**Ordering note:** `publish()` appends every state change to `stateFile`, and `readRuns` keeps the last record per id — so `/status` sees a `running` run within milliseconds of the sha resolving, and the dashboard's poll and the SSE replay both work from the same file. The state file grows by ~8 records per deploy; at a few deploys a day that is negligible, and `deploy/README.md` documents truncation if it ever matters.
